set(APP_SOURCE main.c)
if(PRESENT_MODE_ROTATION_MATRIX)
    set(APP_SOURCE matrix.c)
endif()
if(PRESENT_BOARD STREQUAL "s31-rgb")
    if(NOT IDF_TARGET STREQUAL "esp32s31" OR NOT PRESENT_MODE_ROTATION_MATRIX)
        message(FATAL_ERROR "S31 RGB currently supports the pixel matrix; use benchmark for LVGL")
    endif()
    set(BOARD_SOURCE board_s31_rgb.c)
    set(BOARD_DEPS "")
elseif(IDF_TARGET STREQUAL "esp32s3")
    set(BOARD_SOURCE board_s3.c)
    set(BOARD_DEPS esp_lcd_gc9503 esp_lcd_panel_io_additions esp_io_expander_tca9554 esp_psram)
elseif(IDF_TARGET STREQUAL "esp32p4")
    set(BOARD_SOURCE board_p4.c executor_check.c)
    set(BOARD_DEPS esp_lcd_ek79007 esp_hw_support)
elseif(IDF_TARGET STREQUAL "esp32s31")
    if(PRESENT_NATIVE_ROTATION_MATRIX)
        message(FATAL_ERROR "Native framebuffer matrix requires P4 MIPI or S3 RGB")
    elseif(PRESENT_MODE_ROTATION_MATRIX)
        set(APP_SOURCE matrix.c)
    elseif(PRESENT_S31_TE_VISUAL)
        set(APP_SOURCE visual_main.c)
    else()
        set(APP_SOURCE qspi_main.c)
    endif()
    set(BOARD_SOURCE board_qspi.c)
    set(BOARD_DEPS esp_lcd_co5300 esp_driver_spi efuse esp_mm)
else()
    message(FATAL_ERROR "Only the confirmed S3/P4 boards are supported")
endif()
idf_component_register(SRCS ${APP_SOURCE} ${BOARD_SOURCE} INCLUDE_DIRS "."
    REQUIRES esp_display_present esp_lcd esp_timer esp_psram esp_driver_gpio esp_driver_i2c ${BOARD_DEPS})
idf_component_get_property(_present_dir esp_display_present COMPONENT_DIR)
target_include_directories(${COMPONENT_LIB} PRIVATE "${_present_dir}/src")

if(IDF_TARGET STREQUAL "esp32p4")
    target_compile_definitions(${COMPONENT_LIB} PRIVATE PRESENT_BOARD_EXECUTOR_CHECK=1)
endif()

if(PRESENT_NATIVE_ROTATION_MATRIX)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE PRESENT_NATIVE_ROTATION_MATRIX=1)
    target_compile_options(${COMPONENT_LIB} PRIVATE -Wno-unused-function)
endif()

if(PRESENT_P4_CROSS_AXIS_PROBE)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE PRESENT_P4_CROSS_AXIS_PROBE=1)
endif()

if(PRESENT_P4_TRANSITION_MATRIX)
    target_compile_definitions(${COMPONENT_LIB} PRIVATE PRESENT_P4_TRANSITION_MATRIX=1)
endif()

if(PRESENT_BOARD STREQUAL "s31-rgb")
    target_compile_definitions(${COMPONENT_LIB} PRIVATE PRESENT_BOARD_S31_RGB=1)
endif()
