cmake_minimum_required(VERSION 3.16)
set(ENV{IDF_COMPONENT_MANAGER} "0")
if(NOT PRESENT_DEPS_DIR)
    set(PRESENT_DEPS_DIR "${CMAKE_CURRENT_LIST_DIR}/deps")
endif()
set(EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/../../.." "${PRESENT_DEPS_DIR}")
set(COMPONENTS main)
option(PRESENT_P4_TRANSITION_MATRIX "Public P4 runtime rotation transitions" OFF)
option(PRESENT_P4_CROSS_AXIS_PROBE "Reproduce P4 runtime cross-axis presentation" OFF)
option(PRESENT_MODE_ROTATION_MATRIX "Public presenter mode / rotation matrix" OFF)
option(PRESENT_NATIVE_ROTATION_MATRIX "Native framebuffer rotation and pool matrix" OFF)
if(PRESENT_P4_TRANSITION_MATRIX AND PRESENT_P4_CROSS_AXIS_PROBE)
    message(FATAL_ERROR "Select one P4 rotation diagnostic")
endif()
if(PRESENT_P4_TRANSITION_MATRIX AND NOT PRESENT_MODE_ROTATION_MATRIX)
    message(FATAL_ERROR "PRESENT_P4_TRANSITION_MATRIX requires PRESENT_MODE_ROTATION_MATRIX")
endif()
if(PRESENT_P4_CROSS_AXIS_PROBE AND NOT PRESENT_MODE_ROTATION_MATRIX)
    message(FATAL_ERROR "PRESENT_P4_CROSS_AXIS_PROBE requires PRESENT_MODE_ROTATION_MATRIX")
endif()
if(PRESENT_MODE_ROTATION_MATRIX AND PRESENT_NATIVE_ROTATION_MATRIX)
    message(FATAL_ERROR "Select at most one rotation matrix")
endif()
option(PRESENT_S31_TE_VISUAL "Build the S31 full-frame TE comparison diagnostic" OFF)
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(present_boards)

if(PRESENT_P4_TRANSITION_MATRIX AND NOT IDF_TARGET STREQUAL "esp32p4")
    message(FATAL_ERROR "PRESENT_P4_TRANSITION_MATRIX requires ESP32-P4")
endif()
if(PRESENT_P4_CROSS_AXIS_PROBE AND NOT IDF_TARGET STREQUAL "esp32p4")
    message(FATAL_ERROR "PRESENT_P4_CROSS_AXIS_PROBE requires ESP32-P4")
endif()
if(PRESENT_S31_TE_VISUAL)
    if(NOT IDF_TARGET STREQUAL "esp32s31")
        message(FATAL_ERROR "PRESENT_S31_TE_VISUAL requires the ESP32-S31 board")
    endif()
endif()
if(PRESENT_S31_TE_VISUAL OR PRESENT_MODE_ROTATION_MATRIX)
    target_link_options(${CMAKE_PROJECT_NAME}.elf PRIVATE "-Wl,--wrap=esp_lcd_panel_draw_bitmap")
endif()
