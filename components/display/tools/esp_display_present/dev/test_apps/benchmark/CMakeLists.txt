cmake_minimum_required(VERSION 3.16)
set(ENV{IDF_COMPONENT_MANAGER} "0")
get_filename_component(PRESENT_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
if(NOT PRESENT_BOARD OR NOT PRESENT_CASE_DIR OR NOT PRESENT_LVGL_DIR OR NOT PRESENT_DEPS_DIR)
    message(FATAL_ERROR "Use dev/tools/run.py build --suite benchmark; board, cases and offline dependencies are required")
endif()
if(PRESENT_IMPLEMENTATION STREQUAL "adapter")
    set(IMPLEMENTATION_DIR "$ENV{PRESENT_TEST_ADAPTER_DIR}")
elseif(PRESENT_IMPLEMENTATION STREQUAL "present")
    set(IMPLEMENTATION_DIR "${PRESENT_ROOT}/../esp_lv_present")
else()
    message(FATAL_ERROR "PRESENT_IMPLEMENTATION must be present or adapter")
endif()
set(EXTRA_COMPONENT_DIRS "${PRESENT_ROOT}" "${IMPLEMENTATION_DIR}" "${PRESENT_LVGL_DIR}" "${PRESENT_DEPS_DIR}" "${CMAKE_CURRENT_LIST_DIR}/components")
if(NOT "$ENV{PRESENT_TEST_LCD_DRIVER_DIR}" STREQUAL "")
    list(APPEND EXTRA_COMPONENT_DIRS "$ENV{PRESENT_TEST_LCD_DRIVER_DIR}")
endif()
set(COMPONENTS main)
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(present_benchmark)
if(PRESENT_IMPLEMENTATION STREQUAL "present" AND PRESENT_BOARD STREQUAL "s31-qspi")
    target_link_options(${CMAKE_PROJECT_NAME}.elf PRIVATE "-Wl,--wrap=esp_display_present_te_sync_admit" "-Wl,--wrap=esp_display_present_te_sync_record_tx_start" "-Wl,--wrap=esp_display_present_te_sync_record_tx_done")
endif()

if(PRESENT_PROFILE)
    if(NOT PRESENT_IMPLEMENTATION STREQUAL "present")
        message(FATAL_ERROR "Stage profiling requires present")
    endif()
    foreach(symbol IN ITEMS esp_display_present_engine_begin_next_frame esp_display_present_engine_acquire_buffer esp_display_present_engine_submit_buffer esp_display_present_engine_commit_frame esp_display_present_executor_run esp_cache_msync esp_lcd_panel_draw_bitmap)
        target_link_options(${CMAKE_PROJECT_NAME}.elf PRIVATE "-Wl,--wrap=${symbol}")
    endforeach()
endif()
