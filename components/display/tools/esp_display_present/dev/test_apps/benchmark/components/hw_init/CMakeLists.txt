set(PRESENT_BOARD "$ENV{PRESENT_TEST_PRESENT_BOARD}")
set(PRESENT_IMPLEMENTATION "$ENV{PRESENT_TEST_PRESENT_IMPLEMENTATION}")
set(PRESENT_CASE_DIR "$ENV{PRESENT_TEST_PRESENT_CASE_DIR}")
set(PRESENT_ROOT "$ENV{PRESENT_TEST_PRESENT_ROOT}")
set(BOARD_DIR "${CMAKE_CURRENT_LIST_DIR}/../../../boards/main")
if(PRESENT_BOARD STREQUAL "s3-rgb")
    set(SOURCES "${BOARD_DIR}/board_s3.c" scanout.c)
    set(DEPS esp_lcd_gc9503 esp_lcd_panel_io_additions esp_io_expander_tca9554 esp_driver_i2c esp_psram)
    set(DEFINES BENCH_RGB=1 BENCH_WIDTH=480 BENCH_HEIGHT=480)
elseif(PRESENT_BOARD STREQUAL "s31-rgb")
    set(SOURCES "${BOARD_DIR}/board_s31_rgb.c" scanout.c)
    set(DEPS "")
    set(DEFINES BENCH_RGB=1 BENCH_WIDTH=800 BENCH_HEIGHT=480 PRESENT_BOARD_S31_RGB=1)
elseif(PRESENT_BOARD STREQUAL "p4-dsi")
    set(SOURCES "${BOARD_DIR}/board_p4.c" scanout.c)
    set(DEPS esp_lcd_ek79007 esp_hw_support)
    set(DEFINES BENCH_DSI=1 BENCH_WIDTH=1024 BENCH_HEIGHT=600)
elseif(PRESENT_BOARD STREQUAL "s31-qspi")
    set(SOURCES "${BOARD_DIR}/board_qspi.c" qspi.c)
    set(DEPS esp_lcd_co5300 esp_driver_spi efuse esp_mm)
    set(DEFINES BENCH_QSPI=1 BENCH_WIDTH=480 BENCH_HEIGHT=480)
elseif(PRESENT_BOARD STREQUAL "box3-spi")
    set(SOURCES box3.c)
    set(DEPS esp_lcd_ili9341 esp_driver_spi)
    set(DEFINES BENCH_SPI=1 BENCH_WIDTH=320 BENCH_HEIGHT=240)
else()
    message(FATAL_ERROR "Unknown PRESENT_BOARD")
endif()
idf_component_register(SRCS ${SOURCES} INCLUDE_DIRS . PRIV_INCLUDE_DIRS "${BOARD_DIR}" REQUIRES esp_lcd esp_driver_gpio ${DEPS})
target_compile_definitions(${COMPONENT_LIB} PUBLIC ${DEFINES})

# Benchmark RGB uses real creation facts, including S31 without bounce.
if(PRESENT_BOARD STREQUAL "s3-rgb" OR PRESENT_BOARD STREQUAL "s31-rgb")
    target_sources(${COMPONENT_LIB} PRIVATE "${BOARD_DIR}/board_rgb_info.c")
    target_include_directories(${COMPONENT_LIB} PRIVATE "${PRESENT_ROOT}/src")
    target_compile_definitions(${COMPONENT_LIB} PRIVATE PRESENT_BOARD_RGB_INFO=1)
endif()
if(PRESENT_BOARD STREQUAL "s31-rgb")
    target_compile_definitions(${COMPONENT_LIB} PRIVATE PRESENT_RGB_BOUNCE_LINES=0)
endif()
