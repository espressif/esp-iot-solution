set(PRESENT_BOARD "$ENV{PRESENT_TEST_PRESENT_BOARD}")
set(PRESENT_IMPLEMENTATION "$ENV{PRESENT_TEST_PRESENT_IMPLEMENTATION}")
set(PRESENT_CASE_DIR "$ENV{PRESENT_TEST_PRESENT_CASE_DIR}")
set(PRESENT_ROOT "$ENV{PRESENT_TEST_PRESENT_ROOT}")
if(PRESENT_IMPLEMENTATION STREQUAL "adapter")
    set(BRIDGE bridge_adapter.c)
    set(DEPS esp_lvgl_adapter esp_lcd_touch)
    set(IS_ADAPTER BENCH_ADAPTER=1)
else()
    set(BRIDGE bridge_present.c)
    set(DEPS esp_lv_present)
    set(NATIVE_PRIVATE_INCLUDE "${PRESENT_ROOT}/../esp_lv_present/src")
    if(PRESENT_BOARD STREQUAL "s31-qspi")
        list(APPEND BRIDGE te_meter.c)
    endif()
endif()
if("$ENV{PRESENT_TEST_PROFILE}" STREQUAL "ON")
    list(APPEND BRIDGE profile.c)
    set(PROFILE_DEFINE BENCH_PROFILE=1)
    list(APPEND DEPS esp_mm)
endif()
if("$ENV{PRESENT_TEST_BUFFER_POLICY}" STREQUAL "legacy-physical-50")
    set(BUFFER_POLICY_DEFINE BENCH_BUFFER_LEGACY=1)
endif()
idf_component_register(SRCS main.c buffer_audit.c ${BRIDGE} INCLUDE_DIRS . "${PRESENT_CASE_DIR}" PRIV_INCLUDE_DIRS "${PRESENT_ROOT}/src" ${NATIVE_PRIVATE_INCLUDE} REQUIRES ${DEPS} hw_init lvgl esp_timer)
target_compile_definitions(${COMPONENT_LIB} PRIVATE BENCH_IMPLEMENTATION="${PRESENT_IMPLEMENTATION}" BENCH_BOARD="${PRESENT_BOARD}" ${IS_ADAPTER} ${PROFILE_DEFINE} ${BUFFER_POLICY_DEFINE})

if("$ENV{PRESENT_TEST_DRIVER_TRACE}" STREQUAL "ON")
    target_compile_definitions(${COMPONENT_LIB} PRIVATE BENCH_DRIVER_TRACE=1)
endif()
